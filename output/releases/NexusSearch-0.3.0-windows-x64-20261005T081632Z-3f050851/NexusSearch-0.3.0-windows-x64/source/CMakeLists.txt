cmake_minimum_required(VERSION 3.20)
project(NexusSearch VERSION 0.3.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(NX_MEM_DEBUG   "Guarded allocator: canaries, poisoning, leak registry (tests need it)" ON)
option(NX_WERROR      "Treat warnings as errors" ON)
option(NX_SANITIZE    "Enable ASan+UBSan (Linux/macOS/clang only)" OFF)
option(NX_BUILD_TESTS "Build unit tests" ON)
option(NX_BUILD_BENCH "Build benchmarks" ON)
option(NX_BUILD_SHARED "Also build libnexus as a shared library (for bindings)" ON)
option(NX_TEST_SHARED "Run the same tests through the shared library" ON)

# ---- warnings / baseline ISA --------------------------------------------------
set(NX_COMPILE_OPTIONS "")
if(MSVC)
  list(APPEND NX_COMPILE_OPTIONS /W4 /utf-8 /D_CRT_SECURE_NO_WARNINGS)
  if(NX_WERROR)
    list(APPEND NX_COMPILE_OPTIONS /WX)
  endif()
else()
  list(APPEND NX_COMPILE_OPTIONS -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
  if(NX_WERROR)
    list(APPEND NX_COMPILE_OPTIONS -Werror)
  endif()
  # Baseline x86-64-v2 (SSE4.2 + POPCNT). AVX2/FMA/BMI2 code is compiled with
  # per-function target attributes and selected at runtime (src/core/nx_simd.h).
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    list(APPEND NX_COMPILE_OPTIONS -march=x86-64-v2)
  endif()
  if(MINGW)
    list(APPEND NX_COMPILE_OPTIONS -D__USE_MINGW_ANSI_STDIO=1)
  endif()
  if(NX_SANITIZE)
    list(APPEND NX_COMPILE_OPTIONS -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()

# ---- library --------------------------------------------------------------------
file(GLOB_RECURSE NX_LIB_SOURCES CONFIGURE_DEPENDS
     ${CMAKE_SOURCE_DIR}/src/*.c)
# CLI / server entry points are not part of the library
list(FILTER NX_LIB_SOURCES EXCLUDE REGEX ".*/src/(cli|app)/.*")

add_library(nexus STATIC ${NX_LIB_SOURCES})
target_include_directories(nexus PUBLIC ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/include)
target_compile_options(nexus PRIVATE ${NX_COMPILE_OPTIONS})
if(NX_MEM_DEBUG)
  target_compile_definitions(nexus PUBLIC NX_MEM_DEBUG=1)
endif()
if(NOT MSVC)
  find_package(Threads REQUIRED)
  target_link_libraries(nexus PUBLIC Threads::Threads m)
endif()
if(WIN32)
  target_link_libraries(nexus PUBLIC ws2_32 bcrypt)
endif()
if(NX_SANITIZE AND NOT MSVC)
  target_link_options(nexus PUBLIC -fsanitize=address,undefined)
endif()

if(NX_BUILD_SHARED)
  add_library(nexus_shared SHARED ${NX_LIB_SOURCES})
  set_target_properties(nexus_shared PROPERTIES OUTPUT_NAME nexus POSITION_INDEPENDENT_CODE ON)
  target_include_directories(nexus_shared PUBLIC ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/include)
  target_compile_options(nexus_shared PRIVATE ${NX_COMPILE_OPTIONS})
  target_compile_definitions(nexus_shared PRIVATE NX_BUILD_SHARED=1)
  if(NX_MEM_DEBUG)
    target_compile_definitions(nexus_shared PUBLIC NX_MEM_DEBUG=1)
  endif()
  if(NX_SANITIZE AND NOT MSVC)
    target_link_options(nexus_shared PUBLIC -fsanitize=address,undefined)
  endif()
  if(NOT MSVC)
    target_link_libraries(nexus_shared PUBLIC Threads::Threads m)
    if(MINGW)
      target_link_options(nexus_shared PRIVATE -static-libgcc)
    endif()
  endif()
  if(WIN32)
    target_link_libraries(nexus_shared PUBLIC ws2_32 bcrypt)
  endif()
endif()

# ---- CLI --------------------------------------------------------------------------
file(GLOB NX_CLI_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/cli/*.c)
if(NX_CLI_SOURCES)
  add_executable(nexus_cli ${NX_CLI_SOURCES})
  set_target_properties(nexus_cli PROPERTIES OUTPUT_NAME nexus)
  target_compile_options(nexus_cli PRIVATE ${NX_COMPILE_OPTIONS})
  target_link_libraries(nexus_cli PRIVATE nexus)
  if(MINGW)
    target_link_options(nexus_cli PRIVATE -static -municode)
  endif()
endif()

# ---- tests -------------------------------------------------------------------------
if(NX_BUILD_TESTS)
  enable_testing()
  file(GLOB NX_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/test_*.c)
  foreach(t ${NX_TEST_SOURCES})
    get_filename_component(tname ${t} NAME_WE)
    add_executable(${tname} ${t})
    target_include_directories(${tname} PRIVATE ${CMAKE_SOURCE_DIR}/tests)
    target_compile_options(${tname} PRIVATE ${NX_COMPILE_OPTIONS})
    target_link_libraries(${tname} PRIVATE nexus)
    if(MINGW)
      target_link_options(${tname} PRIVATE -static)
    endif()
    add_test(NAME ${tname} COMMAND ${tname} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
    set_tests_properties(${tname} PROPERTIES TIMEOUT 60)
    if(NX_BUILD_SHARED AND NX_TEST_SHARED)
      add_executable(${tname}_shared ${t})
      target_include_directories(${tname}_shared PRIVATE ${CMAKE_SOURCE_DIR}/tests)
      target_compile_definitions(${tname}_shared PRIVATE NX_USE_SHARED=1)
      target_compile_options(${tname}_shared PRIVATE ${NX_COMPILE_OPTIONS})
      target_link_libraries(${tname}_shared PRIVATE nexus_shared)
      add_test(NAME ${tname}_shared COMMAND ${tname}_shared WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
      set_tests_properties(${tname}_shared PROPERTIES TIMEOUT 60)
    endif()
  endforeach()
endif()

# ---- benchmarks ------------------------------------------------------------------------
if(NX_BUILD_BENCH)
  file(GLOB NX_BENCH_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/bench/bench_*.c)
  foreach(b ${NX_BENCH_SOURCES})
    get_filename_component(bname ${b} NAME_WE)
    add_executable(${bname} ${b})
    target_include_directories(${bname} PRIVATE ${CMAKE_SOURCE_DIR}/tests ${CMAKE_SOURCE_DIR}/bench)
    target_compile_options(${bname} PRIVATE ${NX_COMPILE_OPTIONS})
    target_link_libraries(${bname} PRIVATE nexus)
    if(MINGW)
      target_link_options(${bname} PRIVATE -static)
    endif()
  endforeach()
endif()
